<S> ::= <QUESTION>
      | <REQUEST>
      | <STATEMENT>

<QUESTION> ::= <WH_QUESTION>
             | <YES_NO_QUESTION>
             | <ALTERNATIVE_QUESTION>
             | <POLITE_QUESTION>

<POLITE_QUESTION> ::= <POLITE_PREFIX> <QUESTION_CORE> <OPT_QUESTION_MARK>
                    | <POLITE_PREFIX> <REQUEST_CORE> <OPT_QUESTION_MARK>

<QUESTION_CORE> ::= <WH_QUESTION>
                  | <YES_NO_QUESTION>
                  | <ALTERNATIVE_QUESTION>

<REQUEST> ::= <REQUEST_CORE> <OPT_QUESTION_MARK>

<STATEMENT> ::= <NP> <VP> <OPT_QUESTION_MARK>

<OPT_QUESTION_MARK> ::= "?"
                       | <EMPTY>

<EMPTY> ::=


# 2. PHRASE STRUCTURE
# NP = noun phrase; VP = verb phrase; PP = prepositional phrase;
# QP = quantity phrase; AdjP = adjective phrase; WHP = wh/interrogative phrase.

<NP> ::= <COURSE_NP>
       | <TOPIC_NP>
       | <ASSIGNMENT_NP>
       | <RESOURCE_NP>
       | <RULE_NP>
       | <WEEK_NP>
       | <GENERAL_NP>

<COURSE_NP> ::= <OPT_COURSE_MARKER> <COURSE_REF>
              | <DETERMINER> <OPT_COURSE_MARKER> <COURSE_REF>
              | <COURSE_REF>

<TOPIC_NP> ::= <OPT_TOPIC_MARKER> <TOPIC>
             | <DETERMINER> <OPT_TOPIC_MARKER> <TOPIC>

<ASSIGNMENT_NP> ::= <ASSIGNMENT_REF>
                  | <DETERMINER> <ASSIGNMENT_REF>

<RESOURCE_NP> ::= <RESOURCE_REF>
                | <DETERMINER> <RESOURCE_REF>

<RULE_NP> ::= <RULE_REF>
            | <DETERMINER> <RULE_REF>

<WEEK_NP> ::= <WEEK_REF>

<GENERAL_NP> ::= <NOUN>
               | <DETERMINER> <NOUN>
               | <ADJECTIVE> <NOUN>
               | <NOUN> <NOUN>
               | <NP> <PP>

<VP> ::= <VERB>
       | <VERB> <NP>
       | <VERB> <QP>
       | <VERB> <PP>
       | <VERB> <NP> <PP>
       | <COPULA> <NP>
       | <COPULA> <ADJP>
       | <COPULA> <WHP>
       | <AUXILIARY> <VERB>
       | <AUXILIARY> <VERB> <NP>
       | <AUXILIARY> <VERB> <QP>
       | <VERB> <WHP>

<PP> ::= <PREPOSITION> <NP>
       | <PREPOSITION> <TOPIC_NP>
       | <PREPOSITION> <COURSE_NP>

<QP> ::= <QUANTITY_WORD> <NOUN>
       | <QUANTITY_WORD> <NOUN> <OPT_QUANTITY_MODIFIER>

<OPT_QUANTITY_MODIFIER> ::= "tất cả"
                           | "tổng cộng"
                           | <EMPTY>

<ADJP> ::= <ADJECTIVE>
         | <DEGREE_WORD> <ADJECTIVE>

<WHP> ::= <WH_WORD>
        | <WH_WORD> <NOUN>
        | <WH_WORD> <NP>
        | <QUANTITY_WORD> <NOUN>


# 3. QUESTION CONSTRUCTIONS
# Vietnamese permits multiple word orders. These rules cover common course-query forms;
# they do not claim to cover all Vietnamese syntax.

<WH_QUESTION> ::= <NP> <VP> <WHP>
                | <NP> <VP> <OPT_WH_FINAL>
                | <WHP> <VP>
                | <WHP> <NP> <VP>
                | <NP> <WH_PREDICATE>
                | <WH_PREDICATE> <NP>
                | <WH_FRONTED> <NP> <VP>

<WH_PREDICATE> ::= <COPULA> <WHP>
                 | <VERB> <WHP>
                 | <AUXILIARY> <VERB> <WHP>

<WH_FRONTED> ::= <WH_WORD>
               | <QUANTITY_WORD> <NOUN>
               | <WH_WORD> <NOUN>

<OPT_WH_FINAL> ::= "gì"
                  | "nào"
                  | "bao nhiêu"
                  | <EMPTY>

<YES_NO_QUESTION> ::= <NP> <VP> <QUESTION_PARTICLE>
                    | <AUXILIARY> <NP> <VP> <QUESTION_PARTICLE>
                    | <NP> <VERB> <NP> <QUESTION_PARTICLE>

<ALTERNATIVE_QUESTION> ::= <NP> <VP> "hay" <VP>
                          | <NP> <VP> "hay là" <VP>

<REQUEST_CORE> ::= <REQUEST_VERB> <NP>
                 | <REQUEST_VERB> <NP> <PP>
                 | <REQUEST_VERB> <VP>
                 | <REQUEST_VERB> <NP> <WHP>
                 | <REQUEST_VERB> <NP> <VERB> <NP>

<REQUEST_VERB> ::= "liệt kê"
                 | "cho biết"
                 | "giải thích"
                 | "tóm tắt"
                 | "giới thiệu"
                 | "tìm"
                 | "tìm giúp"
                 | "cung cấp"
                 | "nêu"
                 | "so sánh"
